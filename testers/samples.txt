void handleUartImage() {
  while (Serial2.available() > 0) {
    char c = Serial2.read();

    if (c == 'S' && !imageStarted) {
      imageStarted = true;
      imageIndex = 0;
      expectedImageSize = 0;
      Serial.println("📷 Image start detected");
    } else if (imageStarted && imageIndex == 0) {
      // Read 4-byte image length
      expectedImageSize = (uint32_t(c) << 24);
      imageIndex++;
    } else if (imageStarted && imageIndex == 1) {
      expectedImageSize |= (uint32_t(c) << 16);
      imageIndex++;
    } else if (imageStarted && imageIndex == 2) {
      expectedImageSize |= (uint32_t(c) << 8);
      imageIndex++;
    } else if (imageStarted && imageIndex == 3) {
      expectedImageSize |= (uint32_t(c));
      imageIndex++;
      Serial.print("📏 Expected image size: ");
      Serial.print(expectedImageSize);
      Serial.println(" bytes");
    } else if (imageStarted && imageIndex > 3) {
      if (imageIndex - 4 < MAX_IMAGE_SIZE) {
        imageBuffer[imageIndex - 4] = c;
      } else {
        Serial.println("Image buffer overflow!");
        imageStarted = false;
        imageIndex = 0;
        continue;
      }

      if (imageIndex - 4 >= expectedImageSize) {
        // End of image received
        if (c == 'E') {
          imageReceived = true;
          Serial.println("Full image received and stored");
          lastImageTime = millis();
        } else {
          Serial.println("Unexpected end marker");
        }
        imageStarted = false;
        imageIndex = 0;
      }
    }
  }

  // If image is ready, send it over Bluetooth
  if (imageReceived && SerialBT.availableForWrite() > 0) {
    // Send image over Bluetooth (chunked)
    for (int i = 0; i < expectedImageSize; i++) {
      SerialBT.write(imageBuffer[i]);
    }
    SerialBT.write('E'); // End marker
    SerialBT.flush();
    Serial.println("Image sent over Bluetooth");
    imageReceived = false;
  }
}

void handleBluetoothClient() {
  if (SerialBT.available()) {
    String command = SerialBT.readStringUntil('\n');
    command.trim();
    
    if (command.startsWith("move:")) {
      String direction = command.substring(5);
      if (direction == "forward") {
        moveMotor(MOTOR_FORWARD, 180);
        SerialBT.println("OK: Moving forward");
      } else if (direction == "backward") {
        moveMotor(MOTOR_BACKWARD, 180);
        SerialBT.println("OK: Moving backward");
      } else if (direction == "left") {
        moveMotor(MOTOR_LEFT, 180);
        SerialBT.println("OK: Turning left");
      } else if (direction == "right") {
        moveMotor(MOTOR_RIGHT, 180);
        SerialBT.println("OK: Turning right");
      } else if (direction == "stop") {
        moveMotor(0, 0);
        SerialBT.println("OK: Stopped");
      } else {
        SerialBT.println("ERROR: Unknown direction");
      }
    }
    else if (command.startsWith("servo:")) {
      int angle = command.substring(5).toInt();
      if (angle >= 0 && angle <= 180) {
        fixedServo.write(angle);
        turnServo.write(angle);
        SerialBT.print("OK: Servo angle set to ");
        SerialBT.println(angle);
      } else {
        SerialBT.println("ERROR: Angle must be 0-180");
      }
    }
    else if (command.startsWith("buzzer:")) {
      int commaIndex = command.indexOf(',');
      if (commaIndex != -1) {
        int tone = command.substring(5, commaIndex).toInt();
        int duration = command.substring(commaIndex + 1).toInt();
        if (tone > 0 && duration > 0) {
          tone(BUZZER_PIN, tone);
          delay(duration);
          noTone(BUZZER_PIN);
          SerialBT.print("OK: Buzzer played tone ");
          SerialBT.print(tone);
          SerialBT.print(" for ");
          SerialBT.print(duration);
          SerialBT.println("ms");
        } else {
          SerialBT.println("ERROR: Invalid tone or duration");
        }
      } else {
        SerialBT.println("ERROR: Invalid format");
      }
    }
    else if (command == "distance") {
      int distance = readUltrasonic();
      SerialBT.print("OK: Distance: ");
      SerialBT.print(distance);
      SerialBT.println(" cm");
    }
    else if (command == "sensors") {
      readIRSensors();
      SerialBT.print("OK: Left: ");
      SerialBT.print(leftSensorValue);
      SerialBT.print(", Middle: ");
      SerialBT.print(middleSensorValue);
      SerialBT.print(", Right: ");
      SerialBT.println(rightSensorValue);
    }
    else if (command == "shoot") {
      digitalWrite(SHOOT_PIN, HIGH);
      delay(200);
      digitalWrite(SHOOT_PIN, LOW);
      SerialBT.println("OK: Shooting triggered");
    }
    else if (command == "snapshot") {
      // Send command to camera ESP32 to capture
      Serial2.write('C');
      Serial.println("Sent 'C' to camera ESP32");
      SerialBT.println("OK: Snapshot requested");
    }
    else {
      SerialBT.println("ERROR: Unknown command");
    }
  }
}

void readIRSensors() {
  leftSensorValue = analogRead(LEFT_SENSOR);
  middleSensorValue = analogRead(MIDDLE_SENSOR);
  rightSensorValue = analogRead(RIGHT_SENSOR);
}

int readUltrasonic() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  return duration * 0.034 / 2;
}

void moveMotor(int direction, int speed) {
  // Ensure speed is within valid range
  speed = constrain(speed, 0, 255);
  
  // Set PWM speed
  analogWrite(PWM1_PIN, speed);
  
  // Enable motor control
  digitalWrite(EN_PIN, LOW);
  
  // Latch low
  digitalWrite(STCP_PIN, LOW);
  
  // Send direction data through shift register
  shiftOut(DATA_PIN, SHCP_PIN, MSBFIRST, direction);
  
  // Latch high to update
  digitalWrite(STCP_PIN, HIGH);
}
